Huffman Coding and Bit Serialization in JavaScript

This article provides a practical guide to implementing Huffman coding tree generation and bit serialization using JavaScript. You will learn how to calculate character frequencies, construct a binary prefix tree using a priority queue pattern, extract optimal variable-length codes, and serialize those bit sequences into dense binary data using native Uint8Array buffers.

1. Defining the Node Structure

A Huffman tree consists of leaf nodes representing characters and internal nodes representing combined frequencies.

class HuffmanNode {
  constructor(char, freq, left = null, right = null) {
    this.char = char;
    this.freq = freq;
    this.left = left;
    this.right = right;
  }

  isLeaf() {
    return this.left === null && this.right === null;
  }
}

2. Generating the Huffman Tree

To construct the tree, first count the frequency of each unique character in the input string. Then, place all nodes into a priority list sorted by frequency, continuously combining the two lowest-frequency nodes until only the root node remains.

function buildHuffmanTree(text) {
  if (!text) return null;

  // Step 1: Count character frequencies
  const frequencies = new Map();
  for (const char of text) {
    frequencies.set(char, (frequencies.get(char) || 0) + 1);
  }

  // Step 2: Initialize leaf nodes
  const nodes = [];
  for (const [char, freq] of frequencies.entries()) {
    nodes.push(new HuffmanNode(char, freq));
  }

  // Handle single unique character edge case
  if (nodes.length === 1) {
    return new HuffmanNode(null, nodes[0].freq, nodes[0], null);
  }

  // Step 3: Repeatedly combine the two lowest-frequency nodes
  while (nodes.length > 1) {
    nodes.sort((a, b) => a.freq - b.freq);

    const left = nodes.shift();
    const right = nodes.shift();

    const parent = new HuffmanNode(null, left.freq + right.freq, left, right);
    nodes.push(parent);
  }

  return nodes[0];
}

3. Generating Prefix Bit Codes

Once the tree is built, traverse from the root to each leaf. Assign a 0 bit when moving to the left child and a 1 bit when moving to the right child.

function generateCodes(root) {
  const codes = new Map();

  function traverse(node, currentCode) {
    if (!node) return;

    if (node.isLeaf()) {
      codes.set(node.char, currentCode || "0");
      return;
    }

    traverse(node.left, currentCode + "0");
    traverse(node.right, currentCode + "1");
  }

  traverse(root, "");
  return codes;
}

4. Serializing Bits into Byte Arrays

Huffman codes produce variable-length bit strings (e.g., "0110"). Storing these representations as plain JavaScript strings wastes memory. To achieve real compression, pack these individual bits into an 8-bit typed array (Uint8Array).

function serializeBits(text, codeMap) {
  // Convert full text into a contiguous bit string
  let bitString = "";
  for (const char of text) {
    bitString += codeMap.get(char);
  }

  const totalBits = bitString.length;
  const byteLength = Math.ceil(totalBits / 8);
  const buffer = new Uint8Array(byteLength);

  // Pack every 8 bits into a single byte
  for (let i = 0; i < totalBits; i++) {
    if (bitString[i] === "1") {
      const byteIndex = Math.floor(i / 8);
      const bitOffset = 7 - (i % 8); // Write MSB to LSB
      buffer[byteIndex] |= (1 << bitOffset);
    }
  }

  return { buffer, bitCount: totalBits };
}

5. Complete Usage Example

The following snippet demonstrates the complete workflow from input string to encoded binary buffer:

const input = "abracadabra";

// 1. Build tree and generate codes
const treeRoot = buildHuffmanTree(input);
const codeMap = generateCodes(treeRoot);

console.log("Encoding Map:", Object.fromEntries(codeMap));

// 2. Serialize into binary buffer
const { buffer, bitCount } = serializeBits(input, codeMap);

console.log(`Original Size: ${input.length * 8} bits (${input.length} bytes)`);
console.log(`Compressed Size: ${bitCount} bits (${buffer.byteLength} bytes)`);
console.log("Serialized Bytes:", Array.from(buffer).map(b => b.toString(2).padStart(8, '0')));

For decompression, store both the serialized tree structure (or frequency table) and the total bitCount alongside the buffer to properly identify and discard unused padding bits in the final byte.
